# Optional OCR stack, needed only for PDFs without a text layer (scans).
#
# Installed when the image is built with `--build-arg WITH_PADDLE=true`. It is
# opt-in because PaddlePaddle + PaddleX add well over a gigabyte to the image
# and the recognition models are downloaded on first use.
#
#   docker compose build --build-arg WITH_PADDLE=true ocr-service
paddleocr==3.7.0
# Pinned below 3.3: 3.3.x ships no linux/aarch64 wheel (Docker on Apple
# Silicon), and its oneDNN path fails with "ConvertPirAttribute2RuntimeAttribute
# not support". 3.2.2 has both wheels and works with oneDNN on x86.
paddlepaddle==3.2.2
